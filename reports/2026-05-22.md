WITH grp AS (
  SELECT economic_group_code, SUM(premio_direto) AS premio
  FROM `susep_ses_marts.fct_seguros`
  WHERE periodo_ref BETWEEN DATE '2025-03-01' AND DATE '2026-02-01'
