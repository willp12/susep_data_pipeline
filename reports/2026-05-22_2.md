WITH ult AS (
  SELECT entity_name, economic_group_name,
    SUM(premio_direto) AS pd_ult,

  GROUP BY 1,2

ant AS (
  SELECT entity_name,
    SUM(premio_direto) AS pd_ant
  FROM `susep_ses_marts.fct_seguros`
  WHERE ramo_group_code = '01'
    AND periodo_ref >= DATE_SUB('2026-02-01', INTERVAL 24 MONTH)
    AND periodo_ref <  DATE_SUB('2026-02-01', INTERVAL 12 MONTH)
  GROUP BY 1
)
SELECT
  u.entity_name,
  u.economic_group_name,
  ROUND(u.pd_ult / 1e9, 2) AS premio_direto_bi,
  ROUND(u.pd_ult / SUM(u.pd_ult) OVER () * 100, 1) AS mktshare_pct,
  ROUND(u.sinistro_ult / NULLIF(u.pg_ult, 0) * 100, 1) AS loss_ratio_pct,
  ROUND((u.pd_ult - a.pd_ant) / NULLIF(a.pd_ant, 0) * 100, 1) AS yoy_pct
